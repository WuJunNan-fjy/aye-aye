> **For agentic workers:** 步骤使用 checkbox（`- [ ]`）跟踪。按 Phase 顺序执行；每个 Task 结束必须完成「验证 + 提交」两步。
> 对应设计文档：`docs/2026-09-19-文件工作区与上传整顿设计.md`

**Goal:** 实现「聊天附件本地引用零上传」与「右侧终端/文件面板的三种 SFTP 上传方式（产物直传 / 工作区归档 / 整目录原样）」，底层为分片、断点续传、内存恒定的上传引擎。

**Architecture:** 前端把文件切 5MB 分片以 application/octet-stream 直传后端；后端分片落 `{dataDir}/transfers/{uploadId}/` 暂存并持久化 manifest，complete 时用 SequenceInputStream 顺序拼接、只开一次 SFTP put 流式写远程。Electron 主进程提供本地路径解析、目录选择与 archiver 归档 IPC。新增唯一本地工具 read_local_file，路径受附件授权清单约束。

**Tech Stack:** Spring Boot（Web，octet-stream 直读）、JSch（ChannelSftp + SftpProgressMonitor）、JUnit 5；React 18 + TypeScript + Zustand + axios + xterm；Electron 35（webUtils、dialog、archiver）。

```java
package com.termmind.server.transfer;

/**
 * 分片上传任务清单。序列化为任务目录下的 manifest.json；
 * 后端重启后靠它恢复「已收分片」状态。
 *
 * @param uploadId         客户端确定性生成的任务 id（只允许 [A-Za-z0-9_-]）
 * @param sessionId        SSH 会话 id
 * @param remoteDir        远程目标目录
 * @param name             远程文件名
 * @param size             文件总字节
 * @param totalChunks      分片总数
 * @param chunkSize        常规分片大小（最后一片可小于此值）
 * @param conflictStrategy error(默认) / overwrite / backup
 * @param createdAt        创建时间（epoch millis）
 * @param phase            RECEIVING / COMPLETING / DONE
 */
public record TransferManifest(
        String uploadId,
        String sessionId,
        String remoteDir,
        String name,
        long size,
        int totalChunks,
        int chunkSize,
        String conflictStrategy,
        long createdAt,
        String phase) {

    /** uploadId 只允许安全字符：作为目录名使用，防路径穿越。 */
    public static boolean isSafeUploadId(String s) {
        return s != null && s.length() >= 8 && s.length() <= 128
                && s.matches("[A-Za-z0-9_-]+");
    }

    /** 远程文件名不允许带路径分隔符。 */
    public static boolean isSafeFileName(String s) {
        return s != null && !s.isBlank() && s.length() <= 255
                && !s.contains("/") && !s.contains("\\")
                && !s.contains("\0") && !".".equals(s) && !"..".equals(s);
    }
}
```

```java
    /**
     * 流式上传：调用方提供任意长度 InputStream，只开一次 SFTP put，
     * 内存占用与文件大小无关（GB 级部署包使用）。
     *
     * @param monitor 进度监视器（可为 null）
     */
    public void putStream(String sessionId, String dirPath, String filename,
                          InputStream in, SftpProgressMonitor monitor) throws Exception {
        synchronized (manager.sftpLock(sessionId)) {
            ChannelSftp sftp = manager.getSftp(sessionId);
            ensureDir(sftp, dirPath);
            sftp.put(in, joinPath(dirPath, filename), monitor, ChannelSftp.OVERWRITE);
        }
    }

    /** 简单远程备份：目标存在时 cp 一份到同目录 name.bak-<epochMillis>，返回备份路径；不存在返回 null。 */
    public String backupIfExists(String sessionId, String remoteFilePath) throws Exception {
        SftpATTRS a = stat(sessionId, remoteFilePath);
        if (a == null) return null;
        String bak = remoteFilePath + ".bak-" + System.currentTimeMillis();
        String escaped = remoteFilePath.replace("'", "'\\''");
        String bakEscaped = bak.replace("'", "'\\''");
        SshSessionManager.ExecResult r = manager.exec(sessionId,
                "cp -a '" + escaped + "' '" + bakEscaped + "'", 60);
        if (r.exitCode() != 0) {
            throw new ApiException(500, "远程备份失败：" + r.stderr(), "SFTP_BACKUP_FAILED");
        }
        return bak;
    }
```

```java
package com.termmind.server.transfer;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.termmind.server.ssh.SftpService;
import com.termmind.server.ssh.SshSessionManager;
import com.termmind.server.web.ApiException;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import java.io.IOException;
import java.io.InputStream;
import java.io.SequenceInputStream;
import java.nio.file.FileStore;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Enumeration;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

/**
 * 分片上传服务（方案 C）：分片落本地暂存 → complete 时 SequenceInputStream 直转 SFTP。
 */
@Service
public class TransferService {

    public static final int CHUNK_SIZE = 5 * 1024 * 1024; // 5MB
    private static final long RESERVE_BYTES = 500L * 1024 * 1024;
    private static final long TTL_MILLIS = 24L * 60 * 60 * 1000;
    private static final ObjectMapper MAPPER = new ObjectMapper();

    private final Path root;
    private final SshSessionManager sessions;
    private final SftpService sftp;

    /** 每个 uploadId 一把锁（替代 intern，避免常量池问题）。 */
    private final ConcurrentHashMap<String, Object> locks = new ConcurrentHashMap<>();

    public TransferService(@Value("${termmind.data-dir}") String dataDir,
                           SshSessionManager sessions, SftpService sftp) {
        this.root = Path.of(dataDir, "transfers").toAbsolutePath().normalize();
        this.sessions = sessions;
        this.sftp = sftp;
    }

    public record InitRequest(String uploadId, String sessionId, String remoteDir,
                              String name, long size, String conflictStrategy) {}

    public record InitResponse(String uploadId, List<Integer> receivedChunks,
                               List<Integer> missingChunks, boolean resumed) {}

    // ── init：创建/恢复任务 ──

    public InitResponse init(InitRequest req) throws IOException {
        validateInit(req);
        Object lock = lockFor(req.uploadId());
        synchronized (lock) {
            Path dir = taskDir(req.uploadId());
            TransferManifest existing = readManifest(dir);
            int totalChunks = chunksOf(req.size());

            if (existing != null) {
                return resume(existing, dir);
            }

            // 磁盘空间检查（含余量）
            FileStore store = Files.getFileStore(root);
            long usable = store.getUsableSpace();
            if (usable < req.size() + RESERVE_BYTES) {
                throw new ApiException(507,
                        "磁盘空间不足：需要约 " + (req.size() + RESERVE_BYTES)
                                + " 字节，当前可用 " + usable + " 字节",
                        "INSUFFICIENT_STORAGE");
            }

            Files.createDirectories(dir);
            TransferManifest m = new TransferManifest(
                    req.uploadId(), req.sessionId(), normalizeRemoteDir(req.remoteDir()),
                    req.name(), req.size(), totalChunks, CHUNK_SIZE,
                    req.conflictStrategy() == null || req.conflictStrategy().isBlank()
                            ? "error" : req.conflictStrategy(),
                    System.currentTimeMillis(), "RECEIVING");
            writeManifest(dir, m);
            return new InitResponse(req.uploadId(), List.of(), range(0, totalChunks), false);
        }
    }

    // ── chunk：接收单分片（流式落盘，不进内存） ──

    public void receiveChunk(String uploadId, int index, InputStream body, long contentLength)
            throws IOException {
        if (!TransferManifest.isSafeUploadId(uploadId)) {
            throw new ApiException(400, "非法 uploadId", "BAD_UPLOAD_ID");
        }
        Object lock = lockFor(uploadId);
        synchronized (lock) {
            Path dir = taskDir(uploadId);
            TransferManifest m = readManifest(dir);
            if (m == null) throw new ApiException(404, "任务不存在，请先 init", "NO_TRANSFER");
            if (index < 0 || index >= m.totalChunks()) {
                throw new ApiException(400, "分片序号越界：" + index, "BAD_CHUNK_INDEX");
            }
            long expected = expectedChunkSize(m, index);
            Path tmp = dir.resolve("chunk_" + index + ".tmp");
            Path dst = dir.resolve("chunk_" + index);
            long written = Files.copy(body, tmp, java.nio.file.StandardCopyOption.REPLACE_EXISTING);
            if (written != expected) {
                Files.deleteIfExists(tmp);
                throw new ApiException(400, "分片 " + index + " 大小不符：期望 "
                        + expected + "，实际 " + written, "CHUNK_SIZE_MISMATCH");
            }
            if (contentLength > 0 && contentLength != expected) {
                Files.deleteIfExists(tmp);
                throw new ApiException(400, "Content-Length 与分片大小不符", "CHUNK_LENGTH_MISMATCH");
            }
            // 原子改名：只有完整落盘的分片才可见
            Files.move(tmp, dst, java.nio.file.StandardCopyOption.REPLACE_EXISTING,
                    java.nio.file.StandardCopyOption.ATOMIC_MOVE);
        }
    }

    // ── abort / status ──

    public void abort(String uploadId) throws IOException {
        if (!TransferManifest.isSafeUploadId(uploadId)) {
            throw new ApiException(400, "非法 uploadId", "BAD_UPLOAD_ID");
        }
        synchronized (lockFor(uploadId)) {
            Path dir = taskDir(uploadId);
            if (Files.exists(dir)) deleteRecursively(dir);
        }
    }

    public Map<String, Object> status(String uploadId) throws IOException {
        if (!TransferManifest.isSafeUploadId(uploadId)) {
            throw new ApiException(400, "非法 uploadId", "BAD_UPLOAD_ID");
        }
        synchronized (lockFor(uploadId)) {
            Path dir = taskDir(uploadId);
            TransferManifest m = readManifest(dir);
            Map<String, Object> out = new LinkedHashMap<>();
            if (m == null) {
                out.put("exists", false);
                return out;
            }
            InitResponse r = resume(m, dir);
            out.put("exists", true);
            out.put("phase", m.phase());
            out.put("receivedChunks", r.receivedChunks());
            out.put("missingChunks", r.missingChunks());
            out.put("transferred", transferred(uploadId));
            return out;
        }
    }

    // ── complete 在 Task 4 追加 ──

    // ===== 内部工具 =====

    /** 校验分片并构造顺序拼接流（可独立单测；complete 复用）。 */
    SequenceInputStream openAssembledStream(Path dir, TransferManifest m) throws IOException {
        List<Integer> got = presentChunks(dir, m.totalChunks());
        if (got.size() != m.totalChunks()) {
            List<Integer> miss = new ArrayList<>();
            for (int i = 0; i < m.totalChunks(); i++) if (!got.contains(i)) miss.add(i);
            throw new ApiException(409, "分片不完整，缺失序号：" + miss, "CHUNKS_INCOMPLETE");
        }
        // 逐片校验大小
        for (int i = 0; i < m.totalChunks(); i++) {
            long sz = Files.size(dir.resolve("chunk_" + i));
            if (sz != expectedChunkSize(m, i)) {
                throw new ApiException(400, "分片 " + i + " 大小损坏", "CHUNK_CORRUPT");
            }
        }
        List<InputStream> streams = new ArrayList<>();
        boolean ok = false;
        try {
            for (int i = 0; i < m.totalChunks(); i++) {
                streams.add(Files.newInputStream(dir.resolve("chunk_" + i)));
            }
            Enumeration<InputStream> en = Collections.enumeration(streams);
            SequenceInputStream seq = new SequenceInputStream(en);
            ok = true;
            return seq;
        } finally {
            if (!ok) for (InputStream s : streams) s.close();
        }
    }

    private InitResponse resume(TransferManifest m, Path dir) throws IOException {
        List<Integer> got = presentChunks(dir, m.totalChunks());
        List<Integer> miss = new ArrayList<>();
        for (int i = 0; i < m.totalChunks(); i++) if (!got.contains(i)) miss.add(i);
        return new InitResponse(m.uploadId(), got, miss, true);
    }

    private static List<Integer> presentChunks(Path dir, int total) {
        List<Integer> got = new ArrayList<>();
        for (int i = 0; i < total; i++) {
            if (Files.isRegularFile(dir.resolve("chunk_" + i))) got.add(i);
        }
        return got;
    }

    static long expectedChunkSize(TransferManifest m, int index) {
        if (index < m.totalChunks() - 1) return m.chunkSize();
        return m.size() - (long) m.chunkSize() * (m.totalChunks() - 1);
    }

    static int chunksOf(long size) {
        return (int) ((size + CHUNK_SIZE - 1) / CHUNK_SIZE);
    }

    private void validateInit(InitRequest req) {
        if (req == null) throw new ApiException(400, "请求体为空", "BAD_REQUEST");
        if (!TransferManifest.isSafeUploadId(req.uploadId())) {
            throw new ApiException(400, "非法 uploadId", "BAD_UPLOAD_ID");
        }
        if (!sessions.isSessionValid(req.sessionId())) {
            throw new ApiException(400, "SSH 会话不存在或已失效，请重新连接", "SESSION_INVALID");
        }
        if (req.remoteDir() == null || req.remoteDir().isBlank()) {
            throw new ApiException(400, "remoteDir 不能为空", "BAD_REMOTE_DIR");
        }
        if (!TransferManifest.isSafeFileName(req.name())) {
            throw new ApiException(400, "非法文件名", "BAD_FILE_NAME");
        }
        if (req.size() <= 0) throw new ApiException(400, "size 必须为正", "BAD_SIZE");
    }

    private static String normalizeRemoteDir(String d) {
        String s = d.replace('\\', '/').trim();
        if (s.length() > 1) s = s.replaceAll("/+$", "");
        return s;
    }

    private Path taskDir(String uploadId) {
        Path p = root.resolve(uploadId).normalize();
        if (!p.startsWith(root)) {
            throw new ApiException(400, "非法任务路径", "BAD_UPLOAD_ID");
        }
        return p;
    }

    private Object lockFor(String uploadId) {
        return locks.computeIfAbsent(uploadId, k -> new Object());
    }

    static TransferManifest readManifest(Path dir) {
        try {
            Path f = dir.resolve("manifest.json");
            if (!Files.isRegularFile(f)) return null;
            return MAPPER.readValue(f.toFile(), TransferManifest.class);
        } catch (IOException e) {
            return null;
        }
    }

    static void writeManifest(Path dir, TransferManifest m) throws IOException {
        MAPPER.writerWithDefaultPrettyPrinter().writeValue(dir.resolve("manifest.json").toFile(), m);
    }

    private static List<Integer> range(int from, int until) {
        List<Integer> l = new ArrayList<>();
        for (int i = from; i < until; i++) l.add(i);
        return l;
    }

    static void deleteRecursively(Path p) throws IOException {
        if (!Files.exists(p)) return;
        try (var paths = Files.walk(p)) {
            paths.sorted(Comparator.reverseOrder()).forEach(f -> {
                try { Files.deleteIfExists(f); } catch (IOException ignored) { }
            });
        }
    }

    // ── complete 阶段进度（Task 4 写入） ──
    private final ConcurrentHashMap<String, Long> transferredNow = new ConcurrentHashMap<>();
    public long transferred(String uploadId) {
        return transferredNow.getOrDefault(uploadId, -1L);
    }
}
```

```java
    // ── complete：校验 → SequenceInputStream 直转 SFTP ──

    public void complete(String uploadId) throws Exception {
        if (!TransferManifest.isSafeUploadId(uploadId)) {
            throw new ApiException(400, "非法 uploadId", "BAD_UPLOAD_ID");
        }
        synchronized (lockFor(uploadId)) {
            Path dir = taskDir(uploadId);
            TransferManifest m = readManifest(dir);
            if (m == null) throw new ApiException(404, "任务不存在，请先 init", "NO_TRANSFER");
            if ("DONE".equals(m.phase())) return; // 幂等

            String remotePath = joinRemote(m.remoteDir(), m.name());

            // 冲突策略（在写之前处理）
            String strat = m.conflictStrategy() == null ? "error" : m.conflictStrategy();
            if ("error".equals(strat)) {
                if (sftp.stat(m.sessionId(), remotePath) != null) {
                    throw new ApiException(409, "远程文件已存在：" + remotePath
                            + "（如需覆盖请用 overwrite/backup 策略）", "REMOTE_EXISTS");
                }
            } else if ("backup".equals(strat)) {
                sftp.backupIfExists(m.sessionId(), remotePath);
            }

            // 标记 COMPLETING 并持久化：崩溃后重启清扫能识别（见 Task 5）
            writeManifest(dir, withPhase(m, "COMPLETING"));

            SequenceInputStream seq = null;
            boolean success = false;
            try {
                seq = openAssembledStream(dir, m);
                sftp.putStream(m.sessionId(), m.remoteDir(), m.name(), seq, new CountMonitor(uploadId));
                success = true;
            } finally {
                if (seq != null) seq.close();
                if (success) {
                    transferredNow.remove(uploadId);
                    deleteRecursively(dir);
                } else {
                    // 失败：分片保留，phase 回 RECEIVING，允许重试 complete
                    writeManifest(dir, withPhase(m, "RECEIVING"));
                }
            }
        }
    }

    private static TransferManifest withPhase(TransferManifest m, String phase) {
        return new TransferManifest(m.uploadId(), m.sessionId(), m.remoteDir(), m.name(),
                m.size(), m.totalChunks(), m.chunkSize(), m.conflictStrategy(),
                m.createdAt(), phase);
    }

    private static String joinRemote(String dir, String name) {
        return dir.endsWith("/") ? dir + name : dir + "/" + name;
    }

    /** SftpProgressMonitor：把已转发字节数写入内存表供 status 查询。 */
    private final class CountMonitor implements com.jcraft.jsch.SftpProgressMonitor {
        private final String uploadId;
        private long count;

        CountMonitor(String uploadId) { this.uploadId = uploadId; }

        @Override public void init(int op, String src, String dest, long max) { }

        @Override public boolean count(long bytes) {
            count += bytes;
            transferredNow.put(uploadId, count);
            return true;
        }

        @Override public void end() { }
    }
```

```java
    @PostConstruct
    void startupCleanup() throws IOException {
        Files.createDirectories(root);
        try (var dirs = Files.list(root)) {
            for (Path dir : dirs.toList()) {
                if (!Files.isDirectory(dir)) continue;

                boolean gone = m == null
                        || !sessions.isSessionValid(m.sessionId())
                        || System.currentTimeMillis() - m.createdAt() > TTL_MILLIS;

                    deleteRecursively(dir);
                    continue;
```

```ts
    stream.on('data', (c: Buffer) => chunks.push(c))
    stream.on('end', () => resolve(Buffer.concat(chunks)))
    stream.on('error', reject)

 * 归档工作区目录：
 * - 产物写到 os.tmpdir()/termmind-archives/ws-<ts>.zip|tar.gz；
 * - excludes 已在渲染端转成 minimatch glob（如 **/node_modules/**）；
 * - 通过 archive-progress 事件推进度。
 */
ipcMain.handle(
  'termmind:archive-workspace',
  async (event, dir: string, format: 'zip' | 'tar.gz', excludes: string[]) => {
    if (typeof dir !== 'string' || !existsSync(dir)) {
      throw new Error('工作区目录不存在')
    }
    const outDir = join(os.tmpdir(), 'termmind-archives')
    mkdirSync(outDir, { recursive: true })
    const suffix = format === 'zip' ? 'zip' : 'tar.gz'
    const out = join(outDir, `ws-${Date.now()}.${suffix}`)

    return await new Promise<{ path: string; size: number }>((resolve, reject) => {
      const output = createWriteStream(out)
      const archive = archiver(format === 'zip' ? 'zip' : 'tar', {
        zlib: { level: 6 },
        gzip: format === 'tar.gz',
      })

      output.on('close', () => resolve({ path: out, size: archive.pointer() }))
      output.on('error', reject)
      archive.on('error', reject)

      let entries = 0
      archive.on('entry', () => { entries += 1 })
      archive.on('progress', (pr: { fs: { processedBytes: number; totalBytes: number } }) => {
        event.sender.send('termmind:archive-progress', {
          processedBytes: pr.fs.processedBytes,
          totalBytes: pr.fs.totalBytes,
          entries,
        })
      })

      archive.pipe(output)
      // glob 遍历：dot 文件包含、按 excludes 过滤；不把根目录名打进包
      archive.glob('**/*', {
        cwd: dir,
        dot: true,
        ignore: excludes,
      })
      void archive.finalize()
    })
  },
```

```ts
// ── uploadId：djb2 双段，确定性、长度 ≥16 ──

function djb2(s: string): string {
  let h = 5381
  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) >>> 0
  return h.toString(36).padStart(7, '0')
}

export function makeUploadId(seed: string): string {
  const rev = Array.from(seed).reverse().join('')
  return djb2(seed) + djb2(rev)
}

  onProgress?: (serverBytes: number) => void
}

const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms))

  constructor() {
    super('已取消')
    this.name = 'UploadCanceled'
  }

/** 单文件上传主流程。 */
export async function uploadFile(opts: UploadFileOptions): Promise<void> {
  const { sessionId, remoteDir, source, control } = opts
  const conflict = opts.conflictStrategy ?? 'error'
  const mtime = source.file ? source.file.lastModified : 0
  const seed = `${source.localPath ?? source.name}:${source.size}:${mtime}`
  const uploadId = makeUploadId(seed)
  const totalChunks = Math.ceil(source.size / CHUNK_SIZE)

  const init = await initTransfer({
    uploadId, sessionId, remoteDir,
    name: source.name, size: source.size, conflictStrategy: conflict,
  })

  // 已落盘分片对应的字节数（用于进度基数）
  const bytesOfIndex = (i: number) =>
    i < totalChunks - 1 ? CHUNK_SIZE : source.size - CHUNK_SIZE * (totalChunks - 1)
  let confirmed = init.receivedChunks.reduce((s, i) => s + bytesOfIndex(i), 0)
  opts.onProgress?.(confirmed)

  // ── 分片 worker pool ──
  const missing = init.missingChunks
  let cursor = 0

  const readSlice = async (i: number): Promise<Blob> => {
    const start = i * CHUNK_SIZE
    const end = Math.min(source.size, start + CHUNK_SIZE)
    if (source.file) return source.file.slice(start, end)
    if (!source.localPath || !window.electronAPI) {
      throw new Error('缺少本地文件读取能力')
    }
    const buf = await window.electronAPI.readFileRange(source.localPath, start, end)
    return new Blob([buf])
  }

  const sendOne = async (i: number): Promise<void> => {
    let lastErr: unknown
    for (let attempt = 0; attempt < MAX_RETRY; attempt++) {
      if (control.aborted) throw new UploadCanceled()
      try {
        const blob = await readSlice(i)
        await postChunk(uploadId, i, blob)
        confirmed += bytesOfIndex(i)
        opts.onProgress?.(confirmed)
        return
      } catch (e) {
        lastErr = e
        await sleep(500 * (attempt + 1))
      }
    }
    throw lastErr instanceof Error ? lastErr : new Error('分片发送失败')
  }

  const worker = async (): Promise<void> => {
    for (;;) {
      if (control.aborted) throw new UploadCanceled()
      if (control.paused) {
        await sleep(300)
        continue
      }
      const i = missing[cursor++]
      if (i === undefined) return
      await sendOne(i)
    }
  }

  try {
    await Promise.all(Array.from({ length: MAX_WORKERS }, worker))

    // ── complete：后端拼装直转 SFTP；同时轮询转发进度 ──
    const done = completeTransfer(uploadId)
    let settled = false
    done.then(() => { settled = true }).catch(() => { settled = true })
    while (!settled) {
      await sleep(POLL_MS)
      try {
        const st = await transferStatus(uploadId)
        if (typeof st.transferred === 'number' && st.transferred >= 0) {
          opts.onProgress?.(Math.max(confirmed, st.transferred))
        }
      } catch { /* 状态轮询失败不影响主流程 */ }
    }
    await done
  } catch (e) {
    if (e instanceof UploadCanceled) {
      await abortTransfer(uploadId).catch(() => undefined)
      throw e
    }
    throw e
  }
}

/**
 * 整目录上传：
 * @param files 目录内全部文件（webkitRelativePath 保留相对结构）
 */
export async function uploadDirectory(
  sessionId: string,
  remoteBaseDir: string,
  files: File[],
  control: UploadControl,
  hooks: {
    onFileDone?: (done: number, total: number) => void
    onProgress?: (currentFileBytes: number) => void
  },
): Promise<string[]> {
  const failures: string[] = []
  hooks.onFileDone?.(0, files.length)
  for (let fi = 0; fi < files.length; fi++) {
    if (control.aborted) throw new UploadCanceled()
    const f = files[fi]
    const rel = (f.webkitRelativePath || f.name).split('/')
    const name = rel.pop()!
    const subDir = rel.join('/')
    const dir = subDir ? `${remoteBaseDir}/${subDir}` : remoteBaseDir
    try {
      await uploadFile({
        sessionId,
        remoteDir: normalizeDir(dir),
        source: { name, size: f.size, file: f },
        control,
        onProgress: hooks.onProgress,
      })
    } catch (e) {
      if (e instanceof UploadCanceled) throw e
      failures.push(`${f.webkitRelativePath || f.name}（${e instanceof Error ? e.message : '失败'}）`)
    }
    hooks.onFileDone?.(fi + 1, files.length)
  }
  return failures
}

function normalizeDir(d: string): string {
  const s = d.replace('\\', '/').replace(/\/+$/, '')
  return s === '' ? '/' : s
}
```

```ts
import { create } from 'zustand'
import sftpApi from '@/api/sftpApi'
import {
  uploadFile,
  uploadDirectory,
  type ConflictStrategy,
  type UploadControl,
} from '@/api/transferApi'

export type TaskStatus =
  | 'queued'
  | 'uploading'
  | 'paused'
  | 'done'
  | 'error'
  | 'canceled'

export type UploadKind = 'file' | 'archive' | 'directory'

export interface UploadSpec {
  kind: UploadKind
  sessionId: string
  remoteDir: string
  conflictStrategy?: ConflictStrategy
  // file
  file?: File
  // archive（Electron 归档产物：只有本地路径）
  localPath?: string
  name?: string
  size?: number
  // directory
  files?: File[]
}

export interface UploadTask {
  id: string
  name: string
  size: number
  sentBytes: number
  status: TaskStatus
  kind: UploadKind
  error?: string
  filesTotal?: number
  filesDone?: number
  remoteDir: string
}

interface UploadStore {
  tasks: UploadTask[]
  enqueue: (spec: UploadSpec) => string
  pause: (id: string) => void
  resume: (id: string) => void
  cancel: (id: string) => void
  retry: (id: string) => void
  remove: (id: string) => void
}

const MAX_PARALLEL = 2

// 模块级运行时表（不入 React 状态）
const controls = new Map<string, UploadControl>()
const specs = new Map<string, UploadSpec>()
let seq = 0

function nextId(): string {
  seq += 1
  return `up-${Date.now()}-${seq}`
}

function taskBaseName(spec: UploadSpec): string {
  if (spec.kind === 'file') return spec.file!.name
  if (spec.kind === 'archive') return spec.name ?? 'archive'
  const top = new Set(
    (spec.files ?? []).map((f) => f.webkitRelativePath.split('/')[0]),
  )
  return top.size === 1 ? [...top][0] : `${spec.files?.length ?? 0} 个文件`
}

export const useUploadStore = create<UploadStore>((set, get) => {
  const patch = (id: string, p: Partial<UploadTask>) =>
    set((s) => ({
      tasks: s.tasks.map((t) => (t.id === id ? { ...t, ...p } : t)),
    }))

  /** 调度：并行槽有空就启动下一个 queued。 */
  const tick = () => {
    const running = get().tasks.filter(
      (t) => t.status === 'uploading' || t.status === 'paused',
    ).length
    if (running >= MAX_PARALLEL) return
    const next = get().tasks.find((t) => t.status === 'queued')
    if (next) void runTask(next.id)
  }

  const runTask = async (id: string) => {
    const spec = specs.get(id)
    const control = controls.get(id)
    if (!spec || !control) return
    patch(id, { status: 'uploading', error: undefined })
    try {
      if (spec.kind === 'directory') {
        const files = spec.files ?? []
        // 先创建所有子目录（去重、排序保证父目录在前）
        const dirs = [...new Set(
          files.map((f) => {
            const parts = f.webkitRelativePath.split('/')
            parts.pop()
            return parts.join('/')
          }).filter(Boolean),
        )].sort((a, b) => a.length - b.length)
        for (const d of dirs) {
          await sftpApi.mkdir(spec.sessionId, `${spec.remoteDir}/${d}`)
        }
        const failures = await uploadDirectory(
          spec.sessionId, spec.remoteDir, files, control, {
            onFileDone: (done, total) =>
              patch(id, { filesDone: done, filesTotal: total }),
            onProgress: (b) => patch(id, { sentBytes: b }),
          },
        )
        if (failures.length > 0) {
          throw new Error(`${failures.length} 个文件失败：${failures.slice(0, 3).join('；')}`)
        }
      } else if (spec.kind === 'archive') {
        await uploadFile({
          sessionId: spec.sessionId,
          remoteDir: spec.remoteDir,
          source: {
            name: spec.name!,
            size: spec.size!,
            localPath: spec.localPath,
          },
          conflictStrategy: spec.conflictStrategy,
          control,
          onProgress: (b) => patch(id, { sentBytes: b }),
        })
      } else {
        await uploadFile({
          sessionId: spec.sessionId,
          remoteDir: spec.remoteDir,
          source: {
            name: spec.file!.name,
            size: spec.file!.size,
            file: spec.file,
          },
          conflictStrategy: spec.conflictStrategy,
          control,
          onProgress: (b) => patch(id, { sentBytes: b }),
        })
      }
      patch(id, { status: 'done', sentBytes: get().tasks.find((t) => t.id === id)?.size ?? 0 })
    } catch (e) {
      if (e instanceof Error && e.name === 'UploadCanceled') {
        patch(id, { status: 'canceled' })
      } else {
        patch(id, {
          status: 'error',
          error: e instanceof Error ? e.message : '上传失败',
        })
      }
    } finally {
      window.setTimeout(tick, 0)
    }
  }

  return {
    tasks: [],

    enqueue: (spec: UploadSpec) => {
      const id = nextId()
      const name = taskBaseName(spec)
      const size =
        spec.kind === 'file'
          ? spec.file!.size
          : spec.kind === 'archive'
            ? spec.size ?? 0
            : (spec.files ?? []).reduce((s, f) => s + f.size, 0)
      controls.set(id, { paused: false, aborted: false })
      specs.set(id, spec)
      set((s) => ({
        tasks: [...s.tasks, {
          id, name, size, sentBytes: 0,
          status: 'queued', kind: spec.kind,
          filesTotal: spec.kind === 'directory' ? spec.files?.length : undefined,
          filesDone: 0,
          remoteDir: spec.remoteDir,
        }],
      }))
      window.setTimeout(tick, 0)
      return id
    },

    pause: (id) => {
      const c = controls.get(id)
      if (c) c.paused = true
      patch(id, { status: 'paused' })
    },

    resume: (id) => {
      const c = controls.get(id)
      if (c) c.paused = false
      patch(id, { status: 'uploading' })
      window.setTimeout(tick, 0)
    },

    cancel: (id) => {
      const c = controls.get(id)
      if (c) c.aborted = true
      // queued 任务直接标记（worker 从未启动）
      const t = get().tasks.find((x) => x.id === id)
      if (t?.status === 'queued') patch(id, { status: 'canceled' })
    },

    retry: (id) => {
      const spec = specs.get(id)
      if (!spec) return
      controls.set(id, { paused: false, aborted: false })
      patch(id, { status: 'queued', sentBytes: 0, error: undefined })
      window.setTimeout(tick, 0)
    },

    remove: (id) => {
      controls.delete(id)
      specs.delete(id)
      set((s) => ({ tasks: s.tasks.filter((t) => t.id !== id) }))
    },
  }
})
```

创建 `apps/termmind-frontend-v2/src/lib/dropEntries.ts`（T13/T14/T20 共用；放在这里是因为 T13 是最早使用它的任务）：

```ts
import type { DragEvent } from 'react'

/** drop 条目：dir 保留原始 entry（File 无法反查 entry），file 的 entry 为 null。 */
export interface DropEntry {
  kind: 'file' | 'dir'
  file: File
  entry: FileSystemEntry | null
}

/** 从 drop 事件读取条目（用 webkitGetAsEntry 区分文件与目录）。 */
export async function readDroppedEntries(
  e: DragEvent<HTMLElement>,
): Promise<DropEntry[]> {
  const items = Array.from(e.dataTransfer.items ?? [])
  const out: DropEntry[] = []
  for (const item of items) {
    if (item.kind !== 'file') continue
    const entry = item.webkitGetAsEntry?.()
    if (!entry) {
      const f = item.getAsFile()
      if (f) out.push({ kind: 'file', file: f, entry: null })
      continue
    }
    const f = await new Promise<File | null>((resolve) =>
      (entry as FileSystemFileEntry | FileSystemDirectoryEntry).file(
        (file) => resolve(file), () => resolve(null)))
    if (f) {
      out.push({
        kind: entry.isDirectory ? 'dir' : 'file',
        file: f,
        entry,
      })
    }
  }
  return out
}

/**
 * 递归收集目录内文件，并把相对路径写入 webkitRelativePath
 * （上传 worker 依赖该字段批量建目录；OS 拖入的 File 本身无此字段）。
 */
export async function collectDirectoryFiles(
  dirEntry: FileSystemDirectoryEntry,
  topName: string,
): Promise<File[]> {
  const out: File[] = []

  const walk = async (entry: FileSystemDirectoryEntry, relPrefix: string): Promise<void> => {
    const reader = entry.createReader()
    // readEntries 规范上每批最多 100 条，需循环到返回空
    let batch: FileSystemEntry[] = []
    for (;;) {
      const items = await new Promise<FileSystemEntry[]>((resolve, reject) =>
        reader.readEntries(resolve, reject))
      if (items.length === 0) break
      batch = batch.concat(items)
    }
    for (const ent of batch) {
      if (ent.isDirectory) {
        await walk(ent as FileSystemDirectoryEntry, `${relPrefix}${ent.name}/`)
      } else if (ent.isFile) {
        const f = await new Promise<File>((resolve, reject) =>
          (ent as FileSystemFileEntry).file(resolve, reject))
        Object.defineProperty(f, 'webkitRelativePath', {
          value: `${topName}/${relPrefix}${ent.name}`,
          configurable: true,
        })
        out.push(f)
      }
    }
  }

  await walk(dirEntry, '')
  return out
}
```

```java
package com.termmind.server.local;

import java.nio.file.Path;
import java.util.Optional;
import java.util.Set;
import java.util.concurrent.ConcurrentHashMap;

/**
 * 会话级本地路径白名单：每轮对话只显式授权用户附带的路径。
 *
 * <p>文件按精确路径授权；目录按「目录前缀之内」授权（{@link Path#startsWith}）。
 * 授权随 agent 运行结束清除（fail-closed）：历史会话重放不会自动重新授权，
 * 用户在新一轮再次附带文件时重新注册。线程安全，供 worker 线程与工具线程并发访问。
 */
public final class LocalAttachmentRegistry {

    private record Scope(Set<Path> files, Set<Path> dirs) {
        Scope() {
            this(ConcurrentHashMap.newKeySet(), ConcurrentHashMap.newKeySet());
        }
    }

    private final ConcurrentHashMap<String, Scope> scopes = new ConcurrentHashMap<>();

    /** 注册单个文件（精确匹配）。路径非法时静默忽略。 */
    public void registerFile(String sessionId, String rawPath) {
        normalize(rawPath).ifPresent(scope(sessionId).files()::add);
    }

    /** 注册目录前缀：该目录及其子树均可读。 */
    public void registerDirectory(String sessionId, String rawPath) {
        normalize(rawPath).ifPresent(scope(sessionId).dirs()::add);
    }

    /** 判断目标路径是否在该会话授权范围内。 */
    public boolean isAllowed(String sessionId, String rawPath) {
        Scope s = scopes.get(sessionId);
        if (s == null) return false;
        Optional<Path> target = normalize(rawPath);
        if (target.isEmpty()) return false;
        Path p = target.get();
        if (s.files().contains(p)) return true;
        for (Path dir : s.dirs()) {
            if (p.startsWith(dir)) return true;
        }
        return false;
    }

    /** 清除该会话全部授权（agent 运行结束时调用）。 */
    public void clear(String sessionId) {
        if (sessionId != null) scopes.remove(sessionId);
    }

    private Scope scope(String sessionId) {
        return scopes.computeIfAbsent(sessionId, k -> new Scope());
    }

    private static Optional<Path> normalize(String rawPath) {
        if (rawPath == null || rawPath.isBlank()) return Optional.empty();
        try {
            return Optional.of(Path.of(rawPath).toAbsolutePath().normalize());
        } catch (RuntimeException e) {
            return Optional.empty();
        }
    }
}
```

```java
package com.termmind.server.local;

import com.fasterxml.jackson.databind.JsonNode;
import com.termmind.ai.permission.ToolCategory;
import com.termmind.ai.tool.JsonSchema;
import com.termmind.ai.tool.Tool;
import com.termmind.ai.tool.ToolContext;
import com.termmind.server.web.ApiException;

import javax.imageio.ImageIO;
import javax.imageio.ImageReader;
import javax.imageio.stream.ImageInputStream;
import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.DirectoryStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.Iterator;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.Set;

/**
 * read_local_file —— 唯一的本机文件读取工具（设计第 3 节）。
 *
 * <p>行为：
 * <ul>
 *   <li>READ 类、shouldDefer：经 tool_search 加载，只读免确认；</li>
 *   <li>目标路径必须在本会话 {@link LocalAttachmentRegistry} 授权范围内；</li>
 *   <li>文本文件：返回内容，上限 1MB；二进制文件拒绝；</li>
 *   <li>目录：返回受限清单（递归、跳过 node_modules/.git、上限 2000 项）；</li>
 *   <li>图片：返回格式与尺寸等文本元信息（见类末「图像能力说明」）。</li>
 * </ul>
 */
public final class LocalFileReadTool implements Tool {

    /** 文本读取上限 1MB。 */
    private static final int MAX_TEXT_BYTES = 1024 * 1024;
    /** 目录清单条目上限。 */
    private static final int MAX_LIST_ENTRIES = 2000;
    /** 目录递归时整体跳过的目录名（设计点名 node_modules / .git）。 */
    private static final Set<String> SKIP_DIRS = Set.of("node_modules", ".git");
    private static final Set<String> IMAGE_EXTS =
            Set.of(".png", ".jpg", ".jpeg", ".gif", ".webp", ".bmp");

    private final LocalAttachmentRegistry registry;

    public LocalFileReadTool(LocalAttachmentRegistry registry) {
        this.registry = registry;
    }

    @Override
    public String name() { return "read_local_file"; }

    @Override
    public ToolCategory category() { return ToolCategory.READ; }

    @Override
    public String description() {
        return "读取【用户本机】上已作为附件授权的文件（桌面端后端与用户同机）。"
                + "传入路径必须是用户本轮附带的文件或其目录之内的路径，未授权路径一律拒绝。"
                + "传目录路径返回递归文件清单（跳过 node_modules/.git，最多 2000 项）。"
                + "文本文件最多返回 1MB；需要具体内容时再按清单精确读取，不要一次遍历整个项目。"
                + "图片仅返回尺寸/格式元信息，无法直接看图。";
    }

    @Override
    public JsonNode inputSchema() {
        return JsonSchema.object()
                .property("path", JsonSchema.string()
                        .description("本机文件或目录的绝对路径（来自用户附件清单）"))
                .required("path")
                .build();
    }

    @Override
    public boolean shouldDefer() { return true; }

    @Override
    public long timeoutMs() { return 60_000L; }

    @Override
    public String promptSnippet() {
        return "read_local_file — 在附件白名单内读取用户本机文件/列目录（READ 免确认，文本上限 1MB，目录清单上限 2000 项）。";
    }

    @Override
    public List<String> resolveAffectedPaths(JsonNode input, ToolContext ctx) {
        String path = input.path("path").asText("");
        return path.isBlank() ? List.of() : List.of(path);
    }

    @Override
    public Object execute(JsonNode input, ToolContext ctx) {
        String sessionId = ctx.sessionId();
        String raw = input.path("path").asText("");
        if (sessionId == null || sessionId.isBlank()) {
            throw new ApiException(400, "缺少会话标识", "NO_SESSION");
        }
        final Path path;
        try {
            path = Paths.get(raw).toAbsolutePath().normalize();
        } catch (RuntimeException e) {
            throw new ApiException(400, "路径无效：" + raw, "BAD_PATH");
        }

        if (!registry.isAllowed(sessionId, path.toString())) {
            throw new ApiException(403,
                    "路径不在本轮附件授权范围内（文件精确匹配 / 目录前缀匹配）：" + path,
                    "LOCAL_PATH_NOT_ALLOWED");
        }
        if (!Files.exists(path)) {
            throw new ApiException(404, "文件不可用（已被移动或删除）：" + path, "LOCAL_FILE_GONE");
        }
        try {
            if (Files.isDirectory(path)) {
                return listDirectory(path);
            }
            long size = Files.size(path);
            String fileName = path.getFileName().toString().toLowerCase(Locale.ROOT);
            if (IMAGE_EXTS.contains(extOf(fileName))) {
                return describeImage(path, size);
            }
            if (size > MAX_TEXT_BYTES) {
                throw new ApiException(413,
                        "文件超过 1MB 文本上限（" + size + " 字节），请让用户指定需要的片段或改用文件面板下载",
                        "LOCAL_FILE_TOO_LARGE");
            }
            byte[] bytes = Files.readAllBytes(path);
            if (looksBinary(bytes)) {
                throw new ApiException(415, "二进制文件无法以文本读取：" + path, "LOCAL_BINARY");
            }
            return Map.of(
                    "path", path.toString(),
                    "size_bytes", size,
                    "content", new String(bytes, StandardCharsets.UTF_8));
        } catch (ApiException e) {
            throw e;
        } catch (IOException e) {
            throw new ApiException(500, "读取本地文件失败：" + e.getMessage(), "LOCAL_READ_FAILED");
        }
    }

    // ── 目录清单：手工递归以便在 node_modules/.git 处剪枝 ──

    private Map<String, Object> listDirectory(Path root) throws IOException {
        List<String> entries = new ArrayList<>();
        boolean[] truncated = {false};
        collect(root, root, entries, truncated);

        Map<String, Object> out = new LinkedHashMap<>();
        out.put("directory", root.toString());
        out.put("entry_count", entries.size());
        out.put("skipped_dirs", List.copyOf(SKIP_DIRS));
        out.put("truncated", truncated[0]);
        out.put("note", truncated[0] ? "条目超过 2000 已截断，请给出更具体的子目录" : "");
        out.put("entries", entries);
        return out;
    }

    private void collect(Path dir, Path root, List<String> out, boolean[] truncated) throws IOException {
        if (truncated[0]) return;
        List<Path> children = new ArrayList<>();
        try (DirectoryStream<Path> ds = Files.newDirectoryStream(dir)) {
            ds.forEach(children::add);
        }
        children.sort(Comparator.comparing(p ->
                p.getFileName().toString().toLowerCase(Locale.ROOT)));
        for (Path p : children) {
            String name = p.getFileName().toString();
            boolean isDir = Files.isDirectory(p);
            if (isDir && SKIP_DIRS.contains(name)) continue; // 剪枝：整棵跳过
            if (out.size() >= MAX_LIST_ENTRIES) {
                truncated[0] = true;
                return;
            }
            String rel = root.relativize(p).toString().replace('\\', '/');
            out.add(rel + (isDir ? "/" : ""));
            if (isDir) collect(p, root, out, truncated);
            if (truncated[0]) return;
        }
    }

    // ── 图片：返回尺寸/格式（图像能力降级，见类注释末） ──

    private Map<String, Object> describeImage(Path path, long size) {
        Integer width = null;
        Integer height = null;
        String format = null;
        try (ImageInputStream iis = ImageIO.createImageInputStream(path.toFile())) {
            if (iis != null) {
                Iterator<ImageReader> readers = ImageIO.getImageReaders(iis);
                if (readers.hasNext()) {
                    ImageReader reader = readers.next();
                    try {
                        reader.setInput(iis, true);
                        width = reader.getWidth(0);
                        height = reader.getHeight(0);
                        format = reader.getFormatName();
                    } finally {
                        reader.dispose();
                    }
                }
            }
        } catch (IOException ignored) {
            // 无法解码时仅返回基础信息
        }
        Map<String, Object> out = new LinkedHashMap<>();
        out.put("path", path.toString());
        out.put("size_bytes", size);
        out.put("image_format", format);
        out.put("width", width);
        out.put("height", height);
        out.put("note", "当前工具结果通道仅支持文本，无法把图片像素送入模型；以上为可读取的全部图像元信息。");
        return out;
    }

    private static String extOf(String lowerFileName) {
        int i = lowerFileName.lastIndexOf('.');
        return i < 0 ? "" : lowerFileName.substring(i);
    }

    private static boolean looksBinary(byte[] bytes) {
        int n = Math.min(bytes.length, 1024);
        int suspicious = 0;
        for (int i = 0; i < n; i++) {
            int b = bytes[i] & 0xFF;
            if (b == 0) return true;
            if (b < 7 || (b > 13 && b < 32)) suspicious++;
        }
        return suspicious > n / 10;
    }
}
```
